- [ ] **Step 2: 写失败测试**

新建 `tests/runtime/tabs-before-close.test.ts`（风格照 `tests/runtime/tabs-store.test.ts`：静态导入、`beforeEach` 清 sessionStorage + `resetTabs`、`makeTab` helper）：

```ts
import { beforeEach, describe, expect, it, vi } from "vitest";

import { useTabsStore } from "#src/store/tabs";

const HOME = import.meta.env.VITE_BASE_HOME_PATH;
const makeTab = (key: string, label = key) => ({ key, label }) as any;

const setup = () => {
	useTabsStore.getState().addTab(HOME, makeTab(HOME, "Home"));
	useTabsStore.getState().addTab("/a", makeTab("/a"));
	useTabsStore.getState().addTab("/b", makeTab("/b"));
	useTabsStore.getState().setActiveKey("/a");
};

describe("tabs beforeClose 守卫", () => {
	beforeEach(() => {
		sessionStorage.clear();
		useTabsStore.getState().resetTabs();
		vi.restoreAllMocks();
	});

	it("无守卫时行为与同步版一致（回归基线）", async () => {
		setup();
		await useTabsStore.getState().removeTab("/a");
		expect(useTabsStore.getState().openTabs.has("/a")).toBe(false);
		expect(useTabsStore.getState().activeKey).toBe("/b");
	});

	it("守卫返回 false → 取消关闭，页签与 activeKey 不变", async () => {
		setup();
		useTabsStore.getState().addTab("/a", { beforeClose: () => false });
		await useTabsStore.getState().removeTab("/a");
		expect(useTabsStore.getState().openTabs.has("/a")).toBe(true);
		expect(useTabsStore.getState().activeKey).toBe("/a");
	});

	it("守卫返回 Promise<false> 同样取消", async () => {
		setup();
		useTabsStore.getState().addTab("/a", { beforeClose: async () => false });
		await useTabsStore.getState().removeTab("/a");
		expect(useTabsStore.getState().openTabs.has("/a")).toBe(true);
	});

	it("数组守卫顺序执行，任一 false 短路", async () => {
		setup();
		const order: string[] = [];
		useTabsStore.getState().addTab("/a", {
			beforeClose: [
				() => { order.push("first"); return true; },
				() => { order.push("second"); return false; },
				() => { order.push("third"); return true; },
			],
		});
		await useTabsStore.getState().removeTab("/a");
		expect(order).toEqual(["first", "second"]);
		expect(useTabsStore.getState().openTabs.has("/a")).toBe(true);
	});

	it("守卫抛错 → 取消 + console.error", async () => {
		setup();
		const err = vi.spyOn(console, "error").mockImplementation(() => {});
		useTabsStore.getState().addTab("/a", { beforeClose: () => { throw new Error("boom"); } });
		await useTabsStore.getState().removeTab("/a");
		expect(useTabsStore.getState().openTabs.has("/a")).toBe(true);
		expect(err).toHaveBeenCalledOnce();
	});

	it("批量事务性：右侧关闭任一守卫否决则整批取消", async () => {
		setup();
		useTabsStore.getState().addTab("/b", { beforeClose: () => false });
		await useTabsStore.getState().closeRightTabs(HOME);
```

```ts
	set((state) => {
		if (!routePath?.length)
			return state;
		const newTabs = new Map(state.openTabs);
		const prev = newTabs.get(routePath);
		const merged: TabStateType = { ...prev, ...tabProps };

			];
```

4. `guardClose` + 共用关闭路径（模块作用域，吃 `get`）：

```ts
/** 关闭前守卫：待关闭集合里任一标签页声明了 beforeClose 且返回 false / 抛错时，整体中止。 */
const guardClose = async (get: () => TabsStore, keys: string[]): Promise<boolean> => {
	for (const key of keys) {
		const fn = get().openTabs.get(key)?.beforeClose;
		if (fn == null)
			continue;
		const fns = Array.isArray(fn) ? fn : [fn];
		for (const f of fns) {
			try {
				if (!(await f()))
					return false;
			}
			catch (e) {
				// fail-safe：守卫抛错按中止处理，防误关丢内容；留 console.error 排查线索
				console.error("[tabs] beforeClose 守卫抛错，按中止关闭处理", e);
				return false;
			}
		}
	}
	return true;
};
```

5. 五动作（DRY：每个只算 `removed` 集合，然后走同一 `guardAndClose`）：

```ts
/**
 * 共用关闭路径：in-flight 去重 → 守卫 → 精确删除获批集合（TOCTOU 防护：
 * 不对 live state 重算，await 期间新开的 tab 不受影响）→ activeKey 修正。
 */
const guardAndClose = async (removed: string[], fallbackActive: (remaining: string[]) => string): Promise<void> => {
	if (removed.length === 0)
		return;
	for (const k of removed)
		closingKeys.add(k);
	try {
		if (!(await guardClose(get, removed)))
			return;
		set((state) => {
			const toClose = new Set(removed);
			const newTabs = new Map([...state.openTabs].filter(([k]) => !toClose.has(k)));
			if (newTabs.size === state.openTabs.size)
				return state;
			const remaining = [...newTabs.keys()];
			const newActiveKey = (toClose.has(state.activeKey) || !newTabs.has(state.activeKey))
				? fallbackActive(remaining)
				: state.activeKey;
			return { openTabs: newTabs, activeKey: newActiveKey };
		});
	}
	finally {
		for (const k of removed)
			closingKeys.delete(k);
	}
};
```

各动作：

```ts
removeTab: async (routePath: string) => {
	const homePath = import.meta.env.VITE_BASE_HOME_PATH;
	if (!routePath?.length || routePath === homePath || closingKeys.has(routePath))
		return;
	await guardAndClose([routePath], keys => keys.at(-1) ?? homePath);
},
closeRightTabs: async (routePath: string) => {
	if (closingKeys.has(routePath))
		return;
	const keys = [...get().openTabs.keys()];
	const idx = keys.indexOf(routePath);
	if (idx < 0)
		return;
	await guardAndClose(keys.slice(idx + 1), () => routePath);
},
closeLeftTabs: async (routePath: string) => {
	if (closingKeys.has(routePath))
		return;
	const homePath = import.meta.env.VITE_BASE_HOME_PATH;
	const keys = [...get().openTabs.keys()];
	const idx = keys.indexOf(routePath);
	if (idx < 0)
		return;
	await guardAndClose(keys.slice(0, idx).filter(k => k !== homePath), () => routePath);
},
closeOtherTabs: async (routePath: string) => {
	if (closingKeys.has(routePath))
		return;
	const homePath = import.meta.env.VITE_BASE_HOME_PATH;

		[...get().openTabs.keys()].filter(k => k !== homePath && k !== routePath),

},
```
